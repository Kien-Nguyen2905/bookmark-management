// Engine is the HTTP application.
type Engine interface {
        Start() error
        GetApp() *gin.Engine
}

type engine struct {
        app         *gin.Engine
        cfg         *Config
        redisClient *redis.Client
}

func NewEngine(c *Config, redisClient *redis.Client) Engine {
        e := &engine{app: gin.Default(), cfg: c, redisClient: redisClient}
        e.initRoutes()
        return e
}

func (e *engine) Start() error {
        return e.app.Run(fmt.Sprintf(":%d", e.cfg.AppPort))
}

func (e *engine) GetApp() *gin.Engine {
        return e.app
}

func (e *engine) initRoutes() {
        healthCheckRepo := repository.NewHealthCheck(e.redisClient)
        healthCheckSvc := service.NewHealthCheck(e.cfg.AppName, e.cfg.InstanceID, healthCheckRepo)
        healthCheckHandler := handler.NewHealthCheck(healthCheckSvc)

        genPassSvc := service.NewGenPass()
        urlStorageRepo := repository.NewURLStorage(e.redisClient)
        shortenURLSvc := service.NewShortenUrl(urlStorageRepo, genPassSvc)
        shortenHandler := handler.NewShortenLink(shortenURLSvc)

        e.app.GET("/swagger/*any", ginSwagger.WrapHandler(swaggerFiles.Handler))
        e.app.GET("/health-check", healthCheckHandler.Check)
        e.app.POST("/v1/links/shorten", shortenHandler.CreateShortenLink)
}

// @Summary Health check
// @Description Check the service and its connection to Redis
// @Tags health
// @Produce json
// @Success 200 {object} model.HealthCheckResponse
// @Failure 500 {object} map[string]string
// @Router /health-check [get]
func (h *healthCheck) Check(c *gin.Context) {
        resp, err := h.svc.Check(c.Request.Context())
        if err != nil {
                c.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{"error": "Internal Server Error"})
                return
        }

        c.JSON(http.StatusOK, resp)
}

import (
        "net/http"
        "time"

        "bookmark-management/internal/service"

        "github.com/gin-gonic/gin"
)

// ShortenLink is the HTTP handler of the link shortening endpoint.
type ShortenLink interface {
        CreateShortenLink(c *gin.Context)
}

type shortenLink struct {
        svc service.ShortenUrl
}

func NewShortenLink(svc service.ShortenUrl) ShortenLink {
        return &shortenLink{svc: svc}
}

type shortenLinkInput struct {
        // Exp is the lifetime of the link in seconds.
        Exp int    `json:"exp" binding:"required,gt=0" example:"604800"`
        URL string `json:"url" binding:"required,url" example:"https://example.com"`
}

// CreateShortenLink godoc
// @Summary Shorten a link
// @Description Generate a 7-character code for the url and store it in Redis for exp seconds
// @Tags link
// @Accept application/json
// @Produce application/json
// @Param input body shortenLinkInput true "Input required"
// @Success 200 {object} map[string]string
// @Failure 400 {object} map[string]string
// @Failure 500 {object} map[string]string
// @Router /v1/links/shorten [post]
func (h *shortenLink) CreateShortenLink(c *gin.Context) {
        input := &shortenLinkInput{}
        if err := c.ShouldBindJSON(input); err != nil {
                c.AbortWithStatusJSON(http.StatusBadRequest, gin.H{"error": "Invalid Input"})
                return
        }

        exp := time.Duration(input.Exp) * time.Second
        code, err := h.svc.CreateShortenLink(c.Request.Context(), input.URL, exp)
        if err != nil {
                c.AbortWithStatusJSON(http.StatusInternalServerError, gin.H{"error": "Internal Server Error"})
                return
        }

        c.JSON(http.StatusOK, gin.H{"code": code, "message": "Shorten URL generated successfully!"})
}

// NewClient creates a Redis client configured from environment variables.
// A non-empty envPrefix is prepended to the variable names (e.g. CACHE_REDIS_ADDRESS).
func NewClient(envPrefix string) (*redis.Client, error) {
        cfg, err := newConfig(envPrefix)
        if err != nil {
                return nil, err
        }

        rdb := redis.NewClient(&redis.Options{
                Addr:     cfg.Address,
                Password: cfg.Password,
                DB:       cfg.DB,
        })
        return rdb, nil
}

        "errors"

//go:generate mockery --name=GenPass --filename genpass.go --outpkg svc_mocks

// GenPass generates random strings of letters and digits.
type GenPass interface {
        GeneratePassword(length int) (string, error)
}

type genPass struct{}

func NewGenPass() GenPass {
        return &genPass{}
}

const charset = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"

var ErrInvalidLength = errors.New("password length must be between 6 and 64")

func (g *genPass) GeneratePassword(length int) (string, error) {
        if length < 6 || length > 64 {
                return "", ErrInvalidLength
        }

        password := make([]byte, length)

        for i := 0; i < length; i++ {
                n, err := rand.Int(rand.Reader, big.NewInt(int64(len(charset))))
                if err != nil {
                        return "", err
                }
                password[i] = charset[n.Int64()]
        }
        return string(password), nil
}

// HealthCheck reports the status of the service and its Redis connection.
type HealthCheck interface {
        // Check returns an error if Redis cannot be reached.
        Check(ctx context.Context) (model.HealthCheckResponse, error)
}

type healthCheck struct {
        serviceName string
        instanceID  string
        repo        repository.HealthCheck
}

// NewHealthCheck generates a UUID as the instance id when instanceID is empty.
func NewHealthCheck(serviceName, instanceID string, repo repository.HealthCheck) HealthCheck {
        if instanceID == "" {
                instanceID = uuid.NewString()
        }
        return &healthCheck{serviceName: serviceName, instanceID: instanceID, repo: repo}
}

func (h *healthCheck) Check(ctx context.Context) (model.HealthCheckResponse, error) {
        if err := h.repo.Ping(ctx); err != nil {
                return model.HealthCheckResponse{}, err
        }

        return model.HealthCheckResponse{
                Message:     "OK",
                ServiceName: h.serviceName,
                InstanceID:  h.instanceID,
        }, nil
}

//go:generate mockery --name=ShortenUrl --filename shortenurl.go --outpkg svc_mocks

const DefaultCodeLength = 7

// ShortenUrl creates shortened links.
type ShortenUrl interface {
        // CreateShortenLink generates a code for url and stores it for the duration exp.
        CreateShortenLink(ctx context.Context, url string, exp time.Duration) (string, error)
}

type shortenUrl struct {
        r   repository.URLStorage
        gen GenPass
}

func NewShortenUrl(r repository.URLStorage, gen GenPass) ShortenUrl {
        return &shortenUrl{r: r, gen: gen}
}

func (s *shortenUrl) CreateShortenLink(ctx context.Context, url string, exp time.Duration) (string, error) {
        code, err := s.gen.GeneratePassword(DefaultCodeLength)
        if err != nil {
                return "", err
        }

        err = s.r.StoreURL(ctx, code, url, exp)
        if err != nil {
                return "", err
        }

        return code, nil
}
